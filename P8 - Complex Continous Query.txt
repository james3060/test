print(" Exercise with complex continuous queries with logical, relational & temporal operators on multiple streams ")

import time
import random

# Simulated weather data stream (temperature & humidity)
weather_stream = [
    {"timestamp": "10:00 AM", "temperature": 25, "humidity": 60},
    {"timestamp": "10:01 AM", "temperature": 26, "humidity": 62},
    {"timestamp": "10:02 AM", "temperature": 28, "humidity": 58},
    {"timestamp": "10:03 AM", "temperature": 29, "humidity": 55},
    {"timestamp": "10:04 AM", "temperature": 27, "humidity": 65},
]

# Thresholds
TEMP_HIGH = 28
TEMP_DROP = 26
HUMIDITY_HIGH = 63


# Process weather stream
previous_temp = None
for record in weather_stream:
    temp = record["temperature"]
    humidity = record["humidity"]
    timestamp = record["timestamp"]

    print(f"\n Time: {timestamp} | Temperature: {temp}°C | Humidity: {humidity}%")

    # Temperature exceeds threshold
    if temp > TEMP_HIGH:
        print("Heat Alert! Temperature above", TEMP_HIGH, "°C")

    #Humidity exceeds threshold
    if humidity > HUMIDITY_HIGH:
        print("High Humidity Alert! Humidity:", humidity, "%")

    #Temperature drop detection (Temporal Operator)
    if previous_temp is not None and temp < previous_temp:
        print(f"Temperature Dropped! Previous: {previous_temp}°C → Current: {temp}°C")

    # Update previous temperature
    previous_temp = temp
    time.sleep(2)  # Simulate real-time delay
