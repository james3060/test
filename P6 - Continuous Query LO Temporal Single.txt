#Exercise with continuous queries temporal operators on single stream
import time

# Simulated streaming data (temperature readings over time)
temperature_stream = [
    {"id": 1, "temperature": 45.0, "timestamp": "2025-03-11T10:00:00"},
    {"id": 2, "temperature": 47.2, "timestamp": "2025-03-11T10:01:00"},
    {"id": 3, "temperature": 50.5, "timestamp": "2025-03-11T10:02:00"},
    {"id": 4, "temperature": 49.8, "timestamp": "2025-03-11T10:03:00"},
    {"id": 5, "temperature": 53.1, "timestamp": "2025-03-11T10:04:00"},
]

# Variable to store the last seen temperature
previous_temp = None









# Simulating streaming by processing data one record at a time
for record in temperature_stream:
    print("\nIncoming Data:", record)

    # Temporal condition: Detect increasing trend
    if previous_temp is not None and record["temperature"] > previous_temp:
        print("✔ Trend Alert: Temperature is rising! Previous:", previous_temp, "Current:", record["temperature"])

    # Update previous temperature
    previous_temp = record["temperature"]

    # Simulate real-time streaming delay
    time.sleep(2)
