import spacy
from sumy.parsers.plaintext import PlaintextParser
from sumy.summarizers.text_rank import TextRankSummarizer

# Load the spaCy language model
nlp = spacy.load("en_core_web_sm")

# Custom tokenizer class with both to_sentences and to_words methods
class SpaCyTokenizer:
    def __init__(self, nlp_model):
        self.nlp = nlp_model

    def to_sentences(self, text):
        # Tokenize the text into sentences
        doc = self.nlp(text)
        return [sent.text.strip() for sent in doc.sents]

    def to_words(self, sentence):
        # Tokenize the sentence into words
        doc = self.nlp(sentence)
        return [token.text for token in doc if not token.is_punct]  # Exclude punctuation

# Create an instance of the SpaCyTokenizer
tokenizer = SpaCyTokenizer(nlp)

def extract_summary(text, num_sentences=3):
    # Create a PlaintextParser with the custom tokenizer
    parser = PlaintextParser.from_string(text, tokenizer)
    
    # Create a TextRank summarizer
    summarizer = TextRankSummarizer()
    
    # Generate the summary by extracting the top N sentences
    summary = summarizer(parser.document, num_sentences)
    
    # Combine sentences into the final summary
    return ' '.join([str(sentence) for sentence in summary])

# Example text
text = """Text summarization is a technique used to condense a text document into a shorter version, preserving its core information. 
          It has numerous applications, such as summarizing articles, scientific papers, and legal documents. 
          There are two main types of text summarization: extractive and abstractive. 
          Extractive summarization selects sentences directly from the text, while abstractive summarization generates new sentences."""

# Call the function to extract summary
print(extract_summary(text))
